  { name: 'điện thoại', word: 'điện thoại', key: 'dien-thoai', type: 'word' },
  { name: 'xe đạp', word: 'xe đạp', key: 'xe-dap', type: 'word' },
  { name: 'đồng hồ', word: 'đồng hồ', key: 'dong-ho', type: 'word' },
  { name: 'đèn lồng', word: 'đèn lồng', key: 'den-long', type: 'word' },
  { name: 'quả dừa', word: 'quả dừa', key: 'qua-dua-2', type: 'word' },
  { name: 'mất lượt', word: '', key: 'mat-luot', type: 'penalty' },
  { name: 'cây đa', word: 'cây đa', key: 'cay-da', type: 'word' },
  { name: 'cây dù', word: 'cây dù', key: 'cay-du', type: 'word' }
];

const wordItems = ITEMS.filter(item => item.type === 'word');
document.getElementById('totalChallenges').textContent = wordItems.length;

const DEFAULT_IMAGES = {};

let customImages = JSON.parse(localStorage.getItem('game_vong14_images') || '{}');
let soundEnabled = true;
let score = 0;
let currentRotation = 0;
let isSpinning = false;
let playedIndices = [];
let currentItem = null;
let foundInCurrentTurn = false;

function getImage(key) {
  return customImages[key] || DEFAULT_IMAGES[key] || '';
}

function speak(text) {
  if (!soundEnabled || !('speechSynthesis' in window)) return;
  window.speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(text);
  u.lang = 'vi-VN';
  u.rate = 0.9;
  window.speechSynthesis.speak(u);
}

function showScreen(screenId) {
  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
  document.getElementById(screenId).classList.add('active');
}

document.getElementById('startBtn').addEventListener('click', () => {
  score = 0;
  playedIndices = [];
  document.getElementById('score').textContent = score;
  showScreen('gameScreen');
  speak("Chúng mình cùng quay vòng quay kỳ diệu nào!");
});

const wheelEl = document.getElementById('wheel');
const spinBtn = document.getElementById('spinBtn');

spinBtn.addEventListener('click', () => {
  if (isSpinning) return;
  if (playedIndices.length >= ITEMS.length) {
    endGame();
    return;
  }
  
  isSpinning = true;
  spinBtn.disabled = true;
  document.getElementById('nextBtn').classList.add('hidden');
  document.getElementById('feedback').textContent = '';
  document.getElementById('feedback').className = 'feedback';

  let unplayed = ITEMS.map((_, i) => i).filter(i => !playedIndices.includes(i));
  let chosenIdx = unplayed[Math.floor(Math.random() * unplayed.length)];
  playedIndices.push(chosenIdx);
  currentItem = ITEMS[chosenIdx];

  const totalSectors = ITEMS.length;
  const angleStep = 360 / totalSectors;
  
  const initialAngle = chosenIdx * angleStep + angleStep / 2;
  const targetAngleMod = (360 - initialAngle) % 360;
  
  const currentMod = currentRotation % 360;
  let diff = targetAngleMod - currentMod;
  if (diff < 0) diff += 360;

  const extraRotations = 6 * 360;
  currentRotation = currentRotation + extraRotations + diff;

  wheelEl.style.transform = `rotate(${currentRotation}deg)`;

  setTimeout(() => {
    isSpinning = false;
    spinBtn.disabled = false;
    handleWheelResult(currentItem);
  }, 4000);
});

function handleWheelResult(item) {
  if (item.type === 'bonus') {
    score++;
    document.getElementById('stageText').textContent = `Ô đặc biệt: THÊM LƯỢT!`;
    const picDiv = document.getElementById('picture');
    picDiv.innerHTML = `<div class="missing" style="font-size:24px;color:var(--red);">🎉 Chúc mừng! Con được THÊM LƯỢT và +1 điểm thưởng! 🎉</div>`;
    document.getElementById('wordTitle').textContent = 'Thêm Lượt!';
    document.getElementById('prompt').innerHTML = `Con thật may mắn! Nhấn <b>"Quay tiếp"</b> để tiếp tục nhé.`;
    document.getElementById('letters').innerHTML = '';
    const fb = document.getElementById('feedback');
    fb.textContent = 'Tuyệt vời! Con nhận được phần thưởng đặc biệt!';
    fb.className = 'feedback good';
    document.getElementById('score').textContent = score;
    speak('Chúc mừng con được thêm lượt!');
    document.getElementById('nextBtn').textContent = 'quay tiếp';
    document.getElementById('nextBtn').classList.remove('hidden');
  } else if (item.type === 'penalty') {
    document.getElementById('stageText').textContent = `Ô đặc biệt: MẤT LƯỢT!`;
    const picDiv = document.getElementById('picture');
    picDiv.innerHTML = `<div class="missing" style="font-size:24px;color:var(--red);">😅 Ôi, con bị MẤT LƯỢT lần này rồi!</div>`;
    document.getElementById('wordTitle').textContent = 'Mất Lượt!';
    document.getElementById('prompt').innerHTML = `Không sao cả, con hãy nhấn <b>"Quay tiếp"</b> để gỡ lại nhé!`;
    document.getElementById('letters').innerHTML = '';
    const fb = document.getElementById('feedback');
    fb.textContent = 'Tiếc quá, chúng mình quay lại lần sau nhé!';
    fb.className = 'feedback help';
    speak('Ôi, mất lượt rồi, con quay tiếp nhé!');
    document.getElementById('nextBtn').textContent = 'quay tiếp';
    document.getElementById('nextBtn').classList.remove('hidden');
  } else {
    setupChallenge(item);
  }
}

function setupChallenge(item) {
  foundInCurrentTurn = false;
  document.getElementById('stageText').textContent = `bé hãy tìm chữ đ trong từ: "${item.name}"`;
  
  const picDiv = document.getElementById('picture');
  const imgUrl = getImage(item.key);
  if (imgUrl) {
    picDiv.innerHTML = `<img src="${imgUrl}" alt="${item.name}">`;
  } else {
    picDiv.innerHTML = `<div class="missing">Ô <b>${item.name}</b> trên vòng quay</div>`;
  }

  document.getElementById('wordTitle').textContent = item.word;
  document.getElementById('prompt').innerHTML = `chạm vào đúng chữ <b>đ</b> trong từ ở dưới nhé!`;

  const lettersDiv = document.getElementById('letters');
  lettersDiv.innerHTML = '';

  const chars = Array.from(item.word);
  chars.forEach((char) => {
    const btn = document.createElement('button');
    if (char === ' ') {
      btn.className = 'letter space';
    } else {
      btn.className = 'letter';
      btn.textContent = char;
      btn.addEventListener('click', () => handleLetterClick(btn, char));
    }
    lettersDiv.appendChild(btn);
  });

  speak(`Bé hãy tìm chữ đờ trong từ ${item.name}`);
}

function handleLetterClick(btn, char) {
  if (foundInCurrentTurn) return;

  if (char.toLowerCase() === TARGET_LETTER) {
    btn.classList.add('correct');
    foundInCurrentTurn = true;
    score++;
    document.getElementById('score').textContent = score;

    const fb = document.getElementById('feedback');
    fb.textContent = 'Chính xác! Cô khen bé đã tìm đúng chữ đ!';
    fb.className = 'feedback good';

    speak('Đúng rồi! Cô khen con!');

    if (playedIndices.length >= ITEMS.length) {
      document.getElementById('nextBtn').textContent = 'xem kết quả';
    } else {
      document.getElementById('nextBtn').textContent = 'quay tiếp';
    }
    document.getElementById('nextBtn').classList.remove('hidden');
  } else {
    btn.classList.add('wrong');
    const fb = document.getElementById('feedback');
    fb.textContent = 'Chưa đúng rồi! Con hãy nhìn kỹ và chạm đúng chữ đ nhé!';
    fb.className = 'feedback help';
    speak('Chưa đúng rồi, con thử lại nhé!');
    setTimeout(() => btn.classList.remove('wrong'), 800);
  }
}

});

function endGame() {
  showScreen('finalScreen');
  document.getElementById('finalScore').textContent = `con đã đạt ${score} điểm sau các lượt chơi!`;
  const flowersDiv = document.getElementById('flowers');
  flowersDiv.innerHTML = '';
  const starCount = Math.min(score, 15);
  for (let i = 0; i < starCount; i++) {
    const fl = document.createElement('div');
    fl.className = 'flower';
    flowersDiv.appendChild(fl);
  }
  speak(`Bé xuất sắc hoàn thành trò chơi với số điểm ${score}!`);
}

function renderImageGrid() {
  imageGrid.innerHTML = '';
  ITEMS.forEach(item => {
    const box = document.createElement('div');
    box.className = 'image-set';
    box.innerHTML = `
      <strong>${item.name}</strong>
      <div class="preview"><img src="${getImage(item.key)}" id="prev_${item.key}" alt="${item.name}"></div>
      <input type="file" class="file-input" accept="image/*" data-key="${item.key}">
    `;
    imageGrid.appendChild(box);
  });

  document.querySelectorAll('.file-input').forEach(input => {
    input.addEventListener('change', (e) => {
      const key = e.target.getAttribute('data-key');
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (evt) => {
          customImages[key] = evt.target.result;
          const prevImg = document.getElementById(`prev_${key}`);
          if (prevImg) prevImg.src = evt.target.result;
        };
      reader.readAsDataURL(file);
      }
    });
  });
}
